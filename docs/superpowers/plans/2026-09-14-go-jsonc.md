# go-jsonc Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `github.com/lonhutt/jsonc`, a Go library that parses JSON-with-comments fault-tolerantly with byte offsets, and edits such documents without destroying comments or formatting — full functional parity with microsoft/node-jsonc-parser.

**Architecture:** One scanner is the only code that touches raw text. `Visit` (a SAX-style pass) is built on the scanner and owns all error recovery; `Parse` and `ParseTree` are thin visitors over `Visit` that push and pop a stack. `Format` is a second, independent scanner pass because it needs the trivia that `Visit` discards. `Modify`/`Remove` compose `ParseTree` + `Format` into a splice, emitted as `[]Edit` and applied by `ApplyEdits`.

**Tech Stack:** Go 1.24+, standard library only (`encoding/json`, `unicode/utf16`, `unicode/utf8`, `strings`, `strconv`, `sort`). No third-party dependencies. Tests are stdlib `testing` table tests plus native `go test -fuzz` targets.

**Spec:** `docs/superpowers/specs/2026-09-14-go-jsonc-design.md` — read it before starting. It carries the full API surface, the rationale for the layering, the documented divergences from upstream, and the non-goals.

## Global Constraints

Every task's requirements implicitly include this section.

- **Module path is `github.com/lonhutt/jsonc`.** Package `jsonc`, files flat at the repo root. No subpackages.
- **Standard library only.** Adding any third-party dependency is a plan violation. If a task seems to need one, stop and raise it.
- **Go 1.24 minimum** in `go.mod`. (The spec does not fix a floor; 1.24 is chosen here as the assumption. Nothing in the design needs a newer language feature.)
- **All offsets and lengths are BYTE offsets into the input string**, never runes and never UTF-16 code units. `StartCol` is bytes from the start of the line. This is the single most important invariant in the library — every slice of the input must be byte-indexed.
- **Zero value of `SyntaxKind` is `Unknown`.** `ParseErrorCode` starts at 1; zero is never a valid code.
- **`Parse` and `ParseTree` never return a Go `error`.** They return a best-effort result plus `[]ParseError`. Only `ApplyEdits`, `Modify`, and `Remove` return `error`.
- **Never allocate for tokens that do not need decoding.** `Scanner.Value()` returns a substring slice of the input for numbers and for strings with no escapes.
- **DO NOT RUN GIT COMMANDS.** Do not `git init`, `git add`, `git commit`, `git push`, or create branches, repos, or PRs. Each task ends by reporting that the work is ready and printing a suggested commit message. Lon commits it himself. This overrides the commit step in the writing-plans skill template.
- **TDD is mandatory.** Write the failing test, watch it fail for the right reason, then implement. A test that passes the first time you run it is a broken test — investigate before proceeding.
- **Test data is literal.** Table tests carry the upstream input string and expected output verbatim. No shared builder helpers that obscure which input failed.

```go
// assertKinds scans text to EOF and compares the token sequence.
// Ported from upstream's assertKinds helper in json.test.ts.
func assertKinds(t *testing.T, text string, want ...SyntaxKind) {
	t.Helper()
	s := NewScanner(text)
	var got []SyntaxKind
	for k := s.Scan(); k != EOF; k = s.Scan() {
		got = append(got, k)
	}
	if len(got) != len(want) {
		t.Fatalf("scan(%q) produced %d tokens %v, want %d %v", text, len(got), got, len(want), want)
	}
	for i := range want {
		if got[i] != want[i] {
			t.Errorf("scan(%q) token %d = %v, want %v", text, i, got[i], want[i])
		}
	}
}

func TestSetPositionResetsLineTracking(t *testing.T) {
	// Jumping BACKWARD must not leave stale line state behind. A column is a
	// byte count from the start of a line, so it can never be negative.
	s := NewScanner("{\n}[]")
	s.Scan()
	s.Scan()
	s.Scan()
	s.SetPosition(0)
	if k := s.Scan(); k != OpenBrace {
		t.Fatalf("token = %v, want OpenBrace", k)
	}
	if s.StartLine() != 0 || s.StartColumn() != 0 {
		t.Errorf("after SetPosition(0): line %d col %d, want 0,0", s.StartLine(), s.StartColumn())
	}

	// Jumping FORWARD past a line break must report the real line.
	s2 := NewScanner("{\n\n  []")
	s2.SetPosition(5) // byte 5 is '['; bytes 3 and 4 are the two spaces
	if k := s2.Scan(); k != OpenBracket {
		t.Fatalf("token = %v, want OpenBracket", k)
	}
	if s2.StartLine() != 2 || s2.StartColumn() != 2 {
		t.Errorf("after SetPosition(4): line %d col %d, want 2,2", s2.StartLine(), s2.StartColumn())
	}

	// \r\n counts as one line break across a jump.
	s3 := NewScanner("{\r\n[]")
	s3.SetPosition(3)
	s3.Scan()
	if s3.StartLine() != 1 || s3.StartColumn() != 0 {
		t.Errorf("after SetPosition(3) over CRLF: line %d col %d, want 1,0", s3.StartLine(), s3.StartColumn())
	}
}
```

```go
package jsonc

// SyntaxKind identifies a token produced by Scanner.
type SyntaxKind int

const (
	// Unknown is the zero value: a character the scanner could not classify.
	Unknown SyntaxKind = iota
	OpenBrace
	CloseBrace
	OpenBracket
	CloseBracket
	Comma
	Colon
	NullKeyword
	TrueKeyword
	FalseKeyword
	StringLiteral
	NumericLiteral
	LineComment
	BlockComment
	LineBreak
	Trivia
	EOF
)

var syntaxKindNames = [...]string{
	"Unknown", "OpenBrace", "CloseBrace", "OpenBracket", "CloseBracket",
	"Comma", "Colon", "NullKeyword", "TrueKeyword", "FalseKeyword",
	"StringLiteral", "NumericLiteral", "LineComment", "BlockComment",
	"LineBreak", "Trivia", "EOF",
}

func (k SyntaxKind) String() string {
	if k < 0 || int(k) >= len(syntaxKindNames) {
		return "SyntaxKind(?)"
	}
	return syntaxKindNames[k]
}

// ScanError describes a problem with the token the scanner just produced.
// The scanner still produces a token and a value; it does not stop.
type ScanError int

const (
	ScanNone ScanError = iota
	ScanUnexpectedEndOfComment
	ScanUnexpectedEndOfString
	ScanUnexpectedEndOfNumber
	ScanInvalidUnicode
	ScanInvalidEscapeCharacter
	ScanInvalidCharacter
)

var scanErrorNames = [...]string{
	"None", "UnexpectedEndOfComment", "UnexpectedEndOfString",
	"UnexpectedEndOfNumber", "InvalidUnicode", "InvalidEscapeCharacter",
	"InvalidCharacter",
}

func (e ScanError) String() string {
	if e < 0 || int(e) >= len(scanErrorNames) {
		return "ScanError(?)"
	}
	return scanErrorNames[e]
}

type scannerConfig struct {
	ignoreTrivia bool
}

// ScannerOption configures a Scanner. Construct one with IgnoreTrivia.
type ScannerOption func(*scannerConfig)

// IgnoreTrivia makes Scan skip whitespace, line breaks, and comments.
func IgnoreTrivia() ScannerOption {
	return func(c *scannerConfig) { c.ignoreTrivia = true }
}

// Scanner tokenizes a JSONC document. It is stateful and not safe for
// concurrent use. All offsets are byte offsets into the text.
type Scanner struct {
	text string
	pos  int // byte offset of the next character to read

	token       SyntaxKind
	tokenOffset int
	tokenLength int
	value       string
	err         ScanError

	// Line tracking is incremental: these advance as line terminators are
	// consumed, so StartColumn is O(1) rather than a rescan.
	lineNumber      int
	lineStartOffset int

	tokenLineNumber      int
	tokenLineStartOffset int

	ignoreTrivia bool
}

// NewScanner returns a Scanner positioned before the first token of text.
func NewScanner(text string, opts ...ScannerOption) *Scanner {
	var c scannerConfig
	for _, o := range opts {
		o(&c)
	}
	return &Scanner{text: text, ignoreTrivia: c.ignoreTrivia}
}

func (s *Scanner) Token() SyntaxKind    { return s.token }
func (s *Scanner) Value() string        { return s.value }
func (s *Scanner) Offset() int          { return s.tokenOffset }
func (s *Scanner) Length() int          { return s.tokenLength }
func (s *Scanner) Position() int        { return s.pos }
func (s *Scanner) TokenError() ScanError { return s.err }
func (s *Scanner) StartLine() int       { return s.tokenLineNumber }

// StartColumn is the 0-based byte offset of the token from the start of its line.
func (s *Scanner) StartColumn() int { return s.tokenOffset - s.tokenLineStartOffset }

// SetPosition moves the scanner to a byte offset and resets token state.
//
// Line and column tracking is recomputed for the new position by counting the
// line terminators before it, so StartLine and StartColumn stay absolute and
// correct after a jump. That costs one linear scan of the text up to pos;
// SetPosition is not meant for tight loops over a large document.
//
// node-jsonc-parser leaves its line counters untouched here, which lets a
// backward jump report a stale line and a NEGATIVE column. This port fixes
// that: a column is a byte count from the start of a line and can never be
// negative.
func (s *Scanner) SetPosition(pos int) {
	s.pos = pos
	s.value = ""
	// tokenOffset is pos, not 0: after a jump there is no current token, and
	// reporting the empty token at pos keeps StartColumn non-negative even
	// before the next Scan. node-jsonc-parser uses 0 here.
	s.tokenOffset = pos
	s.tokenLength = 0
	s.token = Unknown
	s.err = ScanNone

	s.lineNumber = 0
	s.lineStartOffset = 0
	for i := 0; i < pos && i < len(s.text); i++ {
		if s.text[i] == '\r' {
			if i+1 < pos && i+1 < len(s.text) && s.text[i+1] == '\n' {
				i++
			}
			s.lineNumber++
			s.lineStartOffset = i + 1
		} else if s.text[i] == '\n' {
			s.lineNumber++
			s.lineStartOffset = i + 1
		}
	}
	s.tokenLineNumber = s.lineNumber
	s.tokenLineStartOffset = s.lineStartOffset
}

// Scan advances to the next token and returns its kind.
func (s *Scanner) Scan() SyntaxKind {
	if s.ignoreTrivia {
		for {
			k := s.scanNext()
			switch k {
			case Trivia, LineBreak, LineComment, BlockComment:
				continue
			default:
				return k
			}
		}
	}
	return s.scanNext()
}

func (s *Scanner) scanNext() SyntaxKind {
	s.value = ""
	s.err = ScanNone
	s.tokenOffset = s.pos
	s.tokenLineNumber = s.lineNumber
	s.tokenLineStartOffset = s.lineStartOffset

	if s.pos >= len(s.text) {
		s.tokenLength = 0
		s.token = EOF
		return EOF
	}

	c := s.text[s.pos]

	switch {
	case isLineBreak(c):
		s.scanLineBreaks()
		s.token = LineBreak
	case isWhitespace(c):
		for s.pos < len(s.text) && isWhitespace(s.text[s.pos]) {
			s.pos++
		}
		s.token = Trivia
	default:
		switch c {
		case '{':
			s.pos++
			s.token = OpenBrace
		case '}':
			s.pos++
			s.token = CloseBrace
		case '[':
			s.pos++
			s.token = OpenBracket
		case ']':
			s.pos++
			s.token = CloseBracket
		case ':':
			s.pos++
			s.token = Colon
		case ',':
			s.pos++
			s.token = Comma
		default:
			// Strings, numbers, keywords, and comments land here until
			// Tasks 2 and 3. Advance by one rune so an unknown multi-byte
			// character cannot split and produce invalid UTF-8 slices.
			s.pos += runeLenAt(s.text, s.pos)
			s.token = Unknown
		}
	}

	s.tokenLength = s.pos - s.tokenOffset
	s.value = s.text[s.tokenOffset:s.pos]
	return s.token
}

// scanLineBreaks consumes a run of line terminators, updating line tracking.
// \r\n counts as one line break.
func (s *Scanner) scanLineBreaks() {
	for s.pos < len(s.text) && isLineBreak(s.text[s.pos]) {
		if s.text[s.pos] == '\r' && s.pos+1 < len(s.text) && s.text[s.pos+1] == '\n' {
			s.pos++
		}
		s.pos++
		s.lineNumber++
		s.lineStartOffset = s.pos
	}
}

func isWhitespace(c byte) bool { return c == ' ' || c == '\t' || c == '\v' || c == '\f' }
func isLineBreak(c byte) bool  { return c == '\n' || c == '\r' }
```

```go
import "unicode/utf8"

// runeLenAt returns the byte length of the rune starting at i, always >= 1 so
// scanning cannot stall on invalid UTF-8.
func runeLenAt(s string, i int) int {
	_, n := utf8.DecodeRuneInString(s[i:])
	if n == 0 {
		return 1
	}
	return n
}
```

```go
import (
	"strings"
	"unicode/utf16"
	"unicode/utf8"

// scanString consumes a quoted string starting at the opening quote and
// returns its decoded value. It sets s.err on malformed input but always
// returns a usable value, matching upstream's behavior of reporting an error
// alongside a result rather than bailing.
func (s *Scanner) scanString() string {
	s.pos++ // opening quote
	start := s.pos
	var b *strings.Builder // allocated only when an escape is found

	for {
		if s.pos >= len(s.text) {
			s.err = ScanUnexpectedEndOfString
			return s.flush(b, start)
		}
		c := s.text[s.pos]
		switch {
		case c == '"':
			out := s.flush(b, start)
			s.pos++ // closing quote
			return out
		case isLineBreak(c):
			// An unescaped line break terminates the string.
			s.err = ScanUnexpectedEndOfString
			return s.flush(b, start)
		case c == '\\':
			if b == nil {
				b = &strings.Builder{}
				b.WriteString(s.text[start:s.pos])
			}
			s.pos++
			if s.pos >= len(s.text) {
				s.err = ScanUnexpectedEndOfString
				return b.String()
			}
			s.writeEscape(b)
			start = s.pos
		default:
			s.pos++
		}
	}
}

// flush returns the builder's contents, or a zero-copy slice of the input when
// no escape was ever encountered.
func (s *Scanner) flush(b *strings.Builder, start int) string {
	if b == nil {
		return s.text[start:s.pos]
	}
	b.WriteString(s.text[start:s.pos])
	return b.String()
}

// writeEscape consumes one escape sequence (s.pos is just past the backslash)
// and writes its value to b.
func (s *Scanner) writeEscape(b *strings.Builder) {
	c := s.text[s.pos]
	s.pos++
	switch c {
	case '"':
		b.WriteByte('"')
	case '\\':
		b.WriteByte('\\')
	case '/':
		b.WriteByte('/')
	case 'b':
		b.WriteByte('\b')
	case 'f':
		b.WriteByte('\f')
	case 'n':
		b.WriteByte('\n')
	case 'r':
		b.WriteByte('\r')
	case 't':
		b.WriteByte('\t')
	case 'u':
		r, ok := s.scanHex4()
		if !ok {
			s.err = ScanInvalidUnicode
			b.WriteRune(utf8.RuneError)
			return
		}
		if utf16.IsSurrogate(rune(r)) {
			// Try to pair it with a following \uXXXX.
			if s.pos+1 < len(s.text) && s.text[s.pos] == '\\' && s.text[s.pos+1] == 'u' {
				save := s.pos
				s.pos += 2
				if r2, ok2 := s.scanHex4(); ok2 {
					if combined := utf16.DecodeRune(rune(r), rune(r2)); combined != utf8.RuneError {
						b.WriteRune(combined)
						return
					}
				}
				s.pos = save
			}
			s.err = ScanInvalidUnicode
			b.WriteRune(utf8.RuneError)
			return
		}
		b.WriteRune(rune(r))
	default:
		s.err = ScanInvalidEscapeCharacter
		b.WriteRune(utf8.RuneError)
	}
}

// scanHex4 reads up to four hex digits, consuming them one at a time and
// stopping at the first non-hex byte or the end of the text. It reports
// success only when it got all four.
//
// Consuming incrementally matters: jumping s.pos to len(s.text) on a shortfall
// would skip past the string's closing quote, so scanString would hit real EOF
// and overwrite ScanInvalidUnicode with ScanUnexpectedEndOfString. Stopping at
// the offending byte leaves the string able to terminate normally and keeps the
// more specific error. This mirrors upstream's scanHexDigits.
func (s *Scanner) scanHex4() (int, bool) {
	v := 0
	digits := 0
	for digits < 4 && s.pos < len(s.text) {
		d := hexVal(s.text[s.pos])
		if d < 0 {
			break
		}
		v = v*16 + d
		s.pos++
		digits++
	}
	if digits < 4 {
		return 0, false
	}
	return v, true
}

func hexVal(c byte) int {
	switch {
	case c >= '0' && c <= '9':
		return int(c - '0')
	case c >= 'a' && c <= 'f':
		return int(c-'a') + 10
	case c >= 'A' && c <= 'F':
		return int(c-'A') + 10
	}
	return -1
}
```

```go
// scanNumber consumes a JSON number and returns its raw source text. It
// returns ok=false when there is no number at all (a bare "-"). A truncated
// number such as "1e" returns the text with s.err set, matching upstream.
func (s *Scanner) scanNumber() (string, bool) {
	start := s.pos
	if s.pos < len(s.text) && s.text[s.pos] == '-' {
		s.pos++
	}
	if s.pos >= len(s.text) || !isDigit(s.text[s.pos]) {
		s.pos = start
		return "", false
	}
	for s.pos < len(s.text) && isDigit(s.text[s.pos]) {
		s.pos++
	}
	if s.pos < len(s.text) && s.text[s.pos] == '.' {
		s.pos++
		if s.pos >= len(s.text) || !isDigit(s.text[s.pos]) {
			s.err = ScanUnexpectedEndOfNumber
			return s.text[start:s.pos], true
		}
		for s.pos < len(s.text) && isDigit(s.text[s.pos]) {
			s.pos++
		}
	}
	if s.pos < len(s.text) && (s.text[s.pos] == 'e' || s.text[s.pos] == 'E') {
```

```go
// scanBlockCommentBody consumes a block comment body (s.pos is just past the
// opening "/*"), updating line tracking for any line terminators inside it.
func (s *Scanner) scanBlockCommentBody() {
	for s.pos < len(s.text) {
		c := s.text[s.pos]
		if c == '*' && s.pos+1 < len(s.text) && s.text[s.pos+1] == '/' {
			s.pos += 2
			return
		}
		if isLineBreak(c) {
			s.scanLineBreaks()
			continue
		}
		s.pos += runeLenAt(s.text, s.pos)
	}
	s.err = ScanUnexpectedEndOfComment
}
```

```go
// ParseError is a single recoverable parse problem. Parsing continues after
// one is reported, so callers receive a slice of them alongside a best-effort
// result. All offsets are byte offsets.
type ParseError struct {

	Offset, Length int
	StartLine      int // 0-based
	StartCol       int // 0-based, bytes from the start of the line

func (e ParseError) Error() string {
	return fmt.Sprintf("%s at offset %d (line %d, column %d)", e.Code, e.Offset, e.StartLine, e.StartCol)
```

```go
type visitState struct {
	s   *Scanner
	v   Visitor
	cfg parseConfig
	// path is the live JSON path stack. Object property names and array
	// indices are written in place as parsing advances.
	path Path
}

// Visit parses text and invokes v's callbacks for each object, array, literal,
// separator, and comment. It reports every recoverable error through
// v.OnError and keeps going.
//
// It returns false only when parsing could not produce a value at all; parse
// errors alone do not make it return false.
func Visit(text string, v Visitor, opts ...ParseOption) bool {
	st := &visitState{
		s:   NewScanner(text),
		v:   v,
		cfg: newParseConfig(opts),
	}

	st.scanNext()
	if st.s.Token() == EOF {
		if st.cfg.allowEmptyContent {
			return true
		}
		st.handleError(ValueExpected, nil, nil)
		return false
	}
	if !st.parseValue() {
		st.handleError(ValueExpected, nil, nil)
		return false
	}
	if st.s.Token() != EOF {
		st.handleError(EndOfFileExpected, nil, nil)
	}
	return true
}

// span builds a Span for the scanner's current token.
func (st *visitState) span() Span {
	// Copy the path stack now; the closure must not observe later mutations.
	snapshot := make(Path, len(st.path))
	copy(snapshot, st.path)
	return Span{
		Offset:    st.s.Offset(),
		Length:    st.s.Length(),
		StartLine: st.s.StartLine(),
		StartCol:  st.s.StartColumn(),
		Path:      func() Path { return snapshot },
	}
}

// scanNext advances past trivia, translating scan errors into parse errors and
// dispatching comments, and returns the next significant token.
func (st *visitState) scanNext() SyntaxKind {
	for {
		token := st.s.Scan()

		switch st.s.TokenError() {
		case ScanInvalidUnicode:
			st.handleError(InvalidUnicode, nil, nil)
		case ScanInvalidEscapeCharacter:
			st.handleError(InvalidEscapeCharacter, nil, nil)
		case ScanUnexpectedEndOfNumber:
			st.handleError(UnexpectedEndOfNumber, nil, nil)
		case ScanUnexpectedEndOfComment:
			st.handleError(UnexpectedEndOfComment, nil, nil)
		case ScanUnexpectedEndOfString:
			st.handleError(UnexpectedEndOfString, nil, nil)
		case ScanInvalidCharacter:
			st.handleError(InvalidCharacter, nil, nil)
		}

		switch token {
		case LineComment, BlockComment:
			if st.cfg.disallowComments {
				st.handleError(InvalidCommentToken, nil, nil)
			} else if st.v.OnComment != nil {
				st.v.OnComment(st.span())
			}
		case Unknown:
			st.handleError(InvalidSymbol, nil, nil)
		case Trivia, LineBreak:
			// skip
		default:
			return token
		}
	}
}

// handleError reports code, then skips tokens to resynchronize. It stops after
// a token in skipUntilAfter (consuming it), or before a token in skipUntil.
func (st *visitState) handleError(code ParseErrorCode, skipUntilAfter, skipUntil []SyntaxKind) {
	if st.v.OnError != nil {
		st.v.OnError(code, st.span())
	}
	if len(skipUntilAfter) == 0 && len(skipUntil) == 0 {
		return
	}
	token := st.s.Token()
	for token != EOF {
		if contains(skipUntilAfter, token) {
			st.scanNext()
			return
		}
		if contains(skipUntil, token) {
			return
		}
		token = st.scanNext()
	}
}

func contains(kinds []SyntaxKind, k SyntaxKind) bool {
	for _, x := range kinds {
		if x == k {
			return true
		}
	}
	return false
}

func (st *visitState) parseValue() bool {
	switch st.s.Token() {
	case OpenBracket:
		return st.parseArray()
	case OpenBrace:
		return st.parseObject()
	case StringLiteral:
		return st.parseLiteral()
	default:
		return st.parseLiteral()
	}
}

func (st *visitState) parseLiteral() bool {
	switch st.s.Token() {
	case NumericLiteral:
		raw := st.s.Value()
		var value any
		if st.cfg.useNumber {
			value = json.Number(raw)
			// Still validate, so InvalidNumberFormat is reported consistently.
			if _, err := strconv.ParseFloat(raw, 64); err != nil {
				st.handleError(InvalidNumberFormat, nil, nil)
				value = json.Number("0")
			}
		} else {
			f, err := strconv.ParseFloat(raw, 64)
			if err != nil {
				st.handleError(InvalidNumberFormat, nil, nil)
				f = 0
			}
			value = f
		}
		if st.v.OnLiteralValue != nil {
			st.v.OnLiteralValue(value, st.span())
		}
	case StringLiteral:
		if st.v.OnLiteralValue != nil {
			st.v.OnLiteralValue(st.s.Value(), st.span())
		}
	case NullKeyword:
		if st.v.OnLiteralValue != nil {
			st.v.OnLiteralValue(nil, st.span())
		}
	case TrueKeyword:
		if st.v.OnLiteralValue != nil {
			st.v.OnLiteralValue(true, st.span())
		}
	case FalseKeyword:
		if st.v.OnLiteralValue != nil {
			st.v.OnLiteralValue(false, st.span())
		}
	default:
		return false
	}
	st.scanNext()
	return true
}

func (st *visitState) separator(sep string) {
	if st.v.OnSeparator != nil {
		st.v.OnSeparator(sep, st.span())
	}
}

func (st *visitState) parseProperty() bool {
	if st.s.Token() != StringLiteral {
		st.handleError(PropertyNameExpected, nil, []SyntaxKind{CloseBrace, Comma})
		return false
	}
	name := st.s.Value()
	// Write the name into the live path before descending into the value.
	st.path[len(st.path)-1] = name
	if st.v.OnObjectProperty != nil {
		st.v.OnObjectProperty(name, st.span())
	}
	st.scanNext()

	if st.s.Token() != Colon {
		st.handleError(ColonExpected, nil, []SyntaxKind{CloseBrace, Comma})
		return true
	}
	st.separator(":")
	st.scanNext()
	if !st.parseValue() {
		st.handleError(ValueExpected, nil, []SyntaxKind{CloseBrace, Comma})
	}
	return true
}

func (st *visitState) parseObject() bool {
	if st.v.OnObjectBegin != nil {
		st.v.OnObjectBegin(st.span())
	}
	// Push a placeholder; parseProperty overwrites it with each key.
	st.path = append(st.path, nil)
	st.scanNext()

	needsComma := false
	for st.s.Token() != CloseBrace && st.s.Token() != EOF {
		if st.s.Token() == Comma {
			if !needsComma {
				st.handleError(ValueExpected, nil, nil)
			}
			st.separator(",")
			st.scanNext()
			if st.s.Token() == CloseBrace && st.cfg.allowTrailingComma {
				break
			}
		} else if needsComma {
			st.handleError(CommaExpected, nil, nil)
		}
		if !st.parseProperty() {
			st.handleError(ValueExpected, nil, []SyntaxKind{CloseBrace, Comma})
		}
		needsComma = true
	}

	st.path = st.path[:len(st.path)-1]
	if st.v.OnObjectEnd != nil {
		st.v.OnObjectEnd(st.span())
	}
	if st.s.Token() != CloseBrace {
		st.handleError(CloseBraceExpected, []SyntaxKind{CloseBrace}, nil)
	} else {
		st.scanNext()
	}
	return true
}

func (st *visitState) parseArray() bool {
	if st.v.OnArrayBegin != nil {
		st.v.OnArrayBegin(st.span())
	}
	st.path = append(st.path, 0)
	st.scanNext()

	needsComma := false
	for st.s.Token() != CloseBracket && st.s.Token() != EOF {
		if st.s.Token() == Comma {
			if !needsComma {
				st.handleError(ValueExpected, nil, nil)
			}
			st.separator(",")
			st.scanNext()
			if st.s.Token() == CloseBracket && st.cfg.allowTrailingComma {
				break
			}
		} else if needsComma {
			st.handleError(CommaExpected, nil, nil)
		}
		if !st.parseValue() {
			st.handleError(ValueExpected, nil, []SyntaxKind{CloseBracket, Comma})
		}
		// Advance the array index in the live path.
		if i, ok := st.path[len(st.path)-1].(int); ok {
			st.path[len(st.path)-1] = i + 1
		}
		needsComma = true
	}

	st.path = st.path[:len(st.path)-1]
	if st.v.OnArrayEnd != nil {
		st.v.OnArrayEnd(st.span())
	}
	if st.s.Token() != CloseBracket {
		st.handleError(CloseBracketExpected, []SyntaxKind{CloseBracket}, nil)
	} else {
		st.scanNext()
	}
	return true
}
```

```go
// Parse parses text and returns the value it represents, using the same Go
// types as json.Unmarshal into an any: map[string]any, []any, string, bool,
// float64 (or json.Number with UseNumber), and nil.
//
// Parsing is fault tolerant. A best-effort value is always returned alongside
// every recoverable problem found; callers wanting strictness check that the
// error slice is empty. Object key order is not preserved — use ParseTree when
// order matters.
func Parse(text string, opts ...ParseOption) (any, []ParseError) {
	var (
		errs    []ParseError
		result  any
		// stack holds the containers currently open. Each entry is either a
		// *[]any or a map[string]any.
		stack   []any
		keys    []string
		hasRoot bool
	)

	// set stores a completed value into the enclosing container, or into
	// result when there is no enclosing container.
	set := func(v any) {
		if len(stack) == 0 {
			result = v
			hasRoot = true
			return
		}
		switch top := stack[len(stack)-1].(type) {
		case *[]any:
			*top = append(*top, v)
		case map[string]any:
			top[keys[len(keys)-1]] = v
		}
	}

	push := func(container any) {
		stack = append(stack, container)
	}

	pop := func() {
		done := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if arr, ok := done.(*[]any); ok {
			set(*arr)
			return
		}
		set(done)
	}

	Visit(text, Visitor{
		OnObjectBegin: func(Span) {
			push(map[string]any{})
			keys = append(keys, "")
		},
		OnObjectProperty: func(name string, _ Span) {
			keys[len(keys)-1] = name
		},
		OnObjectEnd: func(Span) {
			keys = keys[:len(keys)-1]
			pop()
		},
		OnArrayBegin: func(Span) {
			arr := make([]any, 0)
			push(&arr)
		},
		OnArrayEnd: func(Span) {
			pop()
		},
		OnLiteralValue: func(v any, _ Span) {
			set(v)
		},
		OnError: func(c ParseErrorCode, s Span) {
			errs = append(errs, ParseError{
				Code:      c,
				Offset:    s.Offset,
				Length:    s.Length,
				StartLine: s.StartLine,
				StartCol:  s.StartCol,
			})
		},
	}, opts...)

	if !hasRoot {
		return nil, errs
	}
	return result, errs
}
```

Note the `*[]any` in the stack: appending to a slice can reallocate, so the stack must hold a pointer or the appends are lost. Maps do not need this because a map value is already a reference.

```go
// NodeType is the kind of a Node in the parse tree.
type NodeType string

const (
	NodeObject   NodeType = "object"
	NodeArray    NodeType = "array"
	NodeProperty NodeType = "property"
	NodeString   NodeType = "string"
	NodeNumber   NodeType = "number"
	NodeBoolean  NodeType = "boolean"
	NodeNull     NodeType = "null"
)

// Node is one node of a parsed JSONC document. Offsets and lengths are byte
// offsets into the original text.
//
// A Node tree is immutable once ParseTree returns and is safe for concurrent
// reads, despite the Parent pointers.
//
// A property node has exactly two children when well formed: the key (a
// NodeString) and the value. Its ColonOffset locates the ':' between them.
type Node struct {
	Type NodeType
	// Literal is the value of a scalar node (string, number, boolean, null).
	// It is nil for object, array, and property nodes; use Value for those.
	Literal        any
	Offset, Length int
	// ColonOffset is the byte offset of the ':' in a property node, or -1.
	ColonOffset int
	Parent      *Node
	Children    []*Node
}

// Value materializes this subtree as a Go value, using the same types as
// Parse. It replaces upstream's getNodeValue.
func (n *Node) Value() any {
	if n == nil {
		return nil
	}
	switch n.Type {
	case NodeObject:
		m := make(map[string]any, len(n.Children))
		for _, prop := range n.Children {
			if len(prop.Children) != 2 {
				continue // malformed property from a tolerant parse
			}
			key, ok := prop.Children[0].Literal.(string)
			if !ok {
				continue
			}
			m[key] = prop.Children[1].Value()
		}
		return m
	case NodeArray:
		out := make([]any, 0, len(n.Children))
		for _, c := range n.Children {
			out = append(out, c.Value())
		}
		return out
	case NodeProperty:
		if len(n.Children) == 2 {
			return n.Children[1].Value()
		}
		return nil
	default:
		return n.Literal
	}
}

// Path returns the JSON path from the document root to this node. It replaces
// upstream's getNodePath.
func (n *Node) Path() Path {
	if n == nil || n.Parent == nil {
		return Path{}
	}
	parentPath := n.Parent.Path()
	switch n.Parent.Type {
	case NodeProperty:
		// The key node has the same path as the property itself.
		key, _ := n.Parent.Children[0].Literal.(string)
		return append(parentPath, key)
	case NodeArray:
		for i, c := range n.Parent.Children {
			if c == n {
				return append(parentPath, i)
			}
		}
	}
	return parentPath
}

// ParseTree parses text into a tree that preserves source offsets and object
// key order. Like Parse it is fault tolerant, returning a best-effort tree
// alongside every problem found. It returns nil only when the document
// contains no value at all.
func ParseTree(text string, opts ...ParseOption) (*Node, []ParseError) {
	var (
		root    *Node
		current *Node
		errs    []ParseError
	)

	// attach adds child to current, closing a completed property if that
	// leaves the property with both of its children.
	ensureProperty := func() {
		for current != nil && current.Type == NodeProperty && len(current.Children) == 2 {
			current.Length = current.Children[1].Offset + current.Children[1].Length - current.Offset
			current = current.Parent
		}
	}

	attach := func(n *Node) *Node {
		if current == nil {
			root = n
			return n
		}
		n.Parent = current
		current.Children = append(current.Children, n)
		return n
	}

	newNode := func(t NodeType, s Span) *Node {
		return &Node{Type: t, Offset: s.Offset, Length: s.Length, ColonOffset: -1}
	}

	onValueDone := func() {
		ensureProperty()
	}

	Visit(text, Visitor{
		OnObjectBegin: func(s Span) {
			current = attach(newNode(NodeObject, s))
		},
		OnObjectProperty: func(name string, s Span) {
			prop := attach(newNode(NodeProperty, s))
			current = prop
			key := newNode(NodeString, s)
			key.Literal = name
			key.Parent = prop
			prop.Children = append(prop.Children, key)
		},
		OnObjectEnd: func(s Span) {
			// s is positioned at the closing brace (or EOF for an
			// unterminated object); Length covers it.
			current.Length = s.Offset + s.Length - current.Offset
			current = current.Parent
			onValueDone()
		},
		OnArrayBegin: func(s Span) {
			current = attach(newNode(NodeArray, s))
		},
		OnArrayEnd: func(s Span) {
			current.Length = s.Offset + s.Length - current.Offset
			current = current.Parent
			onValueDone()
		},
		OnLiteralValue: func(v any, s Span) {
			n := newNode(literalType(v), s)
			n.Literal = v
			attach(n)
			onValueDone()
		},
		OnSeparator: func(sep string, s Span) {
			if sep == ":" && current != nil && current.Type == NodeProperty {
				current.ColonOffset = s.Offset
			}
		},
		OnError: func(c ParseErrorCode, s Span) {
			errs = append(errs, ParseError{
				Code: c, Offset: s.Offset, Length: s.Length,
				StartLine: s.StartLine, StartCol: s.StartCol,
			})
		},
	}, opts...)

	return root, errs
}

func literalType(v any) NodeType {
	switch v.(type) {
	case nil:
		return NodeNull
	case bool:
		return NodeBoolean
	case string:
		return NodeString
	default:
		// float64 or json.Number
		return NodeNumber
	}
}
```

```go
// FindNodeAtLocation returns the node at path, or nil if there is none.
//
// A malformed path element — anything other than a string or an int — yields
// nil, the same as a path that simply does not resolve.
func FindNodeAtLocation(root *Node, path Path) *Node {
	node := root
	for _, seg := range path {
		if node == nil {
			return nil
		}
		switch key := seg.(type) {
		case string:
			if node.Type != NodeObject {
				return nil
			}
			var found *Node
			for _, prop := range node.Children {
				if len(prop.Children) == 2 {
					if k, ok := prop.Children[0].Literal.(string); ok && k == key {
						found = prop.Children[1]
						break
					}
				}
			}
			if found == nil {
				return nil
			}
			node = found
		case int:
			if node.Type != NodeArray || key < 0 || key >= len(node.Children) {
				return nil
			}
			node = node.Children[key]
		default:
			return nil
		}
	}
	return node
}

// FindNodeAtOffset returns the innermost node containing offset, or nil.
//
// With includeRightBound false a node matches offsets in [Offset, Offset+Length);
// with it true the range is inclusive at both ends, which is what an editor
// wants when the caret sits just past a token.
func FindNodeAtOffset(root *Node, offset int, includeRightBound bool) *Node {
	if root == nil {
		return nil
	}
	if !nodeContains(root, offset, includeRightBound) {
		return nil
	}
	node := root
	for {
		var next *Node
		for _, c := range node.Children {
			if nodeContains(c, offset, includeRightBound) {
				next = c
				break
			}
		}
		if next == nil {
			return node
		}
		node = next
	}
}

func nodeContains(n *Node, offset int, includeRightBound bool) bool {
	if offset < n.Offset {
		return false
	}
	end := n.Offset + n.Length
	if includeRightBound {
		return offset <= end
	}
	return offset < end
}
```

```go
// Location describes the position of an offset inside a JSON document: which
// value encloses it, and whether the offset sits on a property name.
type Location struct {
	// PreviousNode is the node ending at or before the offset, when the offset
	// falls inside or just after a node. It is nil otherwise.
	PreviousNode *Node
	// Path is the JSON path of the value at the offset. For a property key,
	// the last element is the key name.
	Path Path
	// IsAtPropertyKey reports whether the offset is on a property name rather
	// than a value.
	IsAtPropertyKey bool
}

// Matches reports whether pattern describes this location's path. A pattern
// element of "*" matches any single segment, and a pattern shorter than the
// path matches as a prefix.
//
// A malformed pattern element — anything other than a string or an int —
// never matches.
func (l Location) Matches(pattern Path) bool {
	if len(pattern) > len(l.Path) {
		return false
	}
	for i, want := range pattern {
		switch w := want.(type) {
		case string:
			if w == "*" {
				continue
			}
			got, ok := l.Path[i].(string)
			if !ok || got != w {
				return false
			}
		case int:
			got, ok := l.Path[i].(int)
			if !ok || got != w {
				return false
			}
		default:
			return false
		}
	}
	return true
}

// GetLocation evaluates the location of offset within text.
//
// It is defined for incomplete and invalid documents — supporting completion
// in a half-typed file is its main purpose — so it never reports errors.
func GetLocation(text string, offset int) Location {
	var (
		loc         Location
		segments    Path
		done        bool
		previous    *Node
		atKey       bool
	)

	// The visitor records the innermost path enclosing offset. Visiting stops
	// contributing once an event begins past the offset.
	setIfEnclosing := func(s Span, isKey bool) {
		if done || s.Offset > offset {
			return
		}
		segments = s.Path()
		atKey = isKey
	}

	root, _ := ParseTree(text)

	Visit(text, Visitor{
		OnObjectBegin: func(s Span) {
			if s.Offset <= offset {
				segments = s.Path()
				atKey = false
			}
		},
		OnObjectProperty: func(name string, s Span) {
			if s.Offset > offset {
				done = true
				return
			}
			p := s.Path()
			// The property's own path already ends with the key name.
			segments = p
			// The offset is on the key when it falls within the key token,
			// including its right bound so a caret at the end of a key counts.
			atKey = offset >= s.Offset && offset <= s.Offset+s.Length
		},
		OnArrayBegin: func(s Span) {
			if s.Offset <= offset {
				segments = s.Path()
				atKey = false
			}
		},
		OnLiteralValue: func(_ any, s Span) {
			if s.Offset > offset {
				done = true
				return
			}
			setIfEnclosing(s, false)
		},
	})

	if root != nil {
		previous = FindNodeAtOffset(root, offset, true)
	}

	loc.Path = segments
	if loc.Path == nil {
		loc.Path = Path{}
	}
	loc.IsAtPropertyKey = atKey
	loc.PreviousNode = previous
	return loc
}
```

```go
package jsonc

import "strings"

// StripComments removes JavaScript-style comments from text, leaving
// everything else — including line breaks inside block comments — untouched,
// so the result stays line-comparable with the input.
//
// Comment syntax inside a string literal is not a comment and is preserved.
func StripComments(text string) string {
	return stripOrReplace(text, 0, false)
}

// ReplaceComments replaces every character of every comment with ch, keeping
// the text exactly the same length so all byte offsets remain valid. Line
// breaks inside block comments are preserved rather than replaced.
func ReplaceComments(text string, ch rune) string {
	return stripOrReplace(text, ch, true)
}

func stripOrReplace(text string, ch rune, replace bool) string {
	s := NewScanner(text)
	var b strings.Builder
	b.Grow(len(text))

	for {
		k := s.Scan()
		if k == EOF {
			break
		}
		start, end := s.Offset(), s.Offset()+s.Length()
		if k != LineComment && k != BlockComment {
			b.WriteString(text[start:end])
			continue
		}
		if !replace {
			// Keep the line breaks a block comment spanned, so line numbers
			// after the comment do not shift.
			for i := start; i < end; i++ {
				if text[i] == '\n' || text[i] == '\r' {
					b.WriteByte(text[i])
				}
			}
			continue
		}
		for i := start; i < end; i++ {
			if text[i] == '\n' || text[i] == '\r' {
				b.WriteByte(text[i])
				continue
			}
			b.WriteRune(ch)
		}
	}
	return b.String()
}
```

```go
package jsonc

import "strings"

// Range is a byte range in a document.
type Range struct {
	Offset, Length int
}

// Edit replaces Length bytes at Offset with Content. An Edit with Length 0 is
// an insertion; one with empty Content is a deletion.
type Edit struct {
	Offset, Length int
	Content        string
}

// FormatOptions controls Format. The zero value is usable: TabSize 0 means 4,
// and an empty EOL means the terminator is detected from the document.
type FormatOptions struct {
	TabSize            int
	InsertSpaces       bool
	EOL                string
	InsertFinalNewline bool
	// KeepLines preserves the author's existing line breaks and blank lines,
	// normalizing only indentation.
	KeepLines bool
}

// DefaultFormatOptions returns 4-space indentation with LF line endings.
func DefaultFormatOptions() FormatOptions {
	return FormatOptions{TabSize: 4, InsertSpaces: true, EOL: "\n"}
}

func (o FormatOptions) tabSize() int {
	if o.TabSize <= 0 {
		return 4
	}
	return o.TabSize
}

func (o FormatOptions) indentUnit() string {
	if o.InsertSpaces {
		return strings.Repeat(" ", o.tabSize())
	}
	return "\t"
}

// eolFor returns the configured EOL, or the first terminator in text, or "\n".
func (o FormatOptions) eolFor(text string) string {
	if o.EOL != "" {
		return o.EOL
	}
	for i := 0; i < len(text); i++ {
		if text[i] == '\r' {
			if i+1 < len(text) && text[i+1] == '\n' {
				return "\r\n"
			}
			return "\r"
		}
		if text[i] == '\n' {
			return "\n"
		}
	}
	return "\n"
}

// formatter holds the state of one Format pass.
type formatter struct {
	text       string
	o          FormatOptions
	eol        string
	indentUnit string

	// Edits outside [rangeStart, rangeEnd) are discarded.
	rangeStart, rangeEnd int
	// formatTextStart is where scanning begins: the start of the line
	// containing rangeStart.
	formatTextStart int

	s                *Scanner
	edits            []Edit
	numberLineBreaks int
	indentLevel      int
	initialIndent    int
}

// Format returns the edits that reformat text. Passing a nil Range formats the
// whole document; otherwise only edits touching the range are returned, with
// the range widened backward to the start of its first line so nesting depth
// is computed correctly.
//
// Only whitespace between tokens is changed. An already-formatted document
// produces no edits.
func Format(text string, r *Range, o FormatOptions) []Edit {
	f := &formatter{
		text:       text,
		o:          o,
		eol:        o.eolFor(text),
		indentUnit: o.indentUnit(),
	}

	if r == nil {
		f.rangeStart, f.rangeEnd = 0, len(text)
		f.formatTextStart = 0
		f.initialIndent = 0
	} else {
		f.rangeStart = r.Offset
		f.rangeEnd = r.Offset + r.Length
		start := f.rangeStart
		for start > 0 && !isLineBreak(text[start-1]) {
			start--
		}
		f.formatTextStart = start
		f.initialIndent = computeIndentLevel(text[start:f.rangeStart], o)
	}

	f.s = NewScanner(text)
	f.s.SetPosition(f.formatTextStart)
	f.run()
	return f.edits
}

// computeIndentLevel counts the leading indentation of line as indent units.
func computeIndentLevel(line string, o FormatOptions) int {
	i, spaces := 0, 0
	for i < len(line) {
		switch line[i] {
		case ' ':
			spaces++
		case '\t':
			spaces += o.tabSize()
		default:
			return spaces / o.tabSize()
		}
		i++
	}
	return spaces / o.tabSize()
}

// addEdit records a replacement of text[start:end] with content, unless the
// document already says exactly that or the edit falls outside the range.
func (f *formatter) addEdit(content string, start, end int) {
	if start > end || start < 0 || end > len(f.text) {
		return
	}
	if start >= f.rangeEnd || end <= f.rangeStart {
		// Entirely outside the requested range. An insertion exactly at
		// rangeStart still counts, hence the <= / >= asymmetry.
		if !(start == end && start >= f.rangeStart && start <= f.rangeEnd) {
			return
		}
	}
	if f.text[start:end] == content {
		return
	}
	f.edits = append(f.edits, Edit{Offset: start, Length: end - start, Content: content})
}

// scanNext advances past trivia, counting the line breaks it crossed.
func (f *formatter) scanNext() SyntaxKind {
	token := f.s.Scan()
	f.numberLineBreaks = 0
	for token == Trivia || token == LineBreak {
		if token == LineBreak {
			f.numberLineBreaks++
		}
		token = f.s.Scan()
	}
	return token
}

func (f *formatter) newLineAndIndent() string {
	return f.eol + strings.Repeat(f.indentUnit, f.initialIndent+f.indentLevel)
}

func (f *formatter) run() {
	f.indentLevel = 0

	firstToken := f.scanNext()
	if firstToken == EOF {
		return
	}

	// Indentation before the very first token.
	firstStart := f.s.Offset()
	f.addEdit(strings.Repeat(f.indentUnit, f.initialIndent), f.formatTextStart, firstStart)

	for firstToken != EOF {
		firstEnd := f.s.Offset() + f.s.Length()
		secondToken := f.scanNext()

		replaceContent := ""
		needsLineBreak := false

		// Comments sharing a line with the previous token keep a single space
		// before them; a line comment forces a break after it.
		for f.numberLineBreaks == 0 && (secondToken == LineComment || secondToken == BlockComment) {
			commentStart := f.s.Offset()
			f.addEdit(" ", firstEnd, commentStart)
			firstEnd = f.s.Offset() + f.s.Length()
			needsLineBreak = secondToken == LineComment
			if needsLineBreak {
				replaceContent = f.newLineAndIndent()
			} else {
				replaceContent = ""
			}
			secondToken = f.scanNext()
		}

		switch secondToken {
		case CloseBrace:
			if firstToken != OpenBrace {
				f.indentLevel--
				replaceContent = f.newLineAndIndent()
			}
		case CloseBracket:
			if firstToken != OpenBracket {
				f.indentLevel--
				replaceContent = f.newLineAndIndent()
			}
		case EOF:
			if f.numberLineBreaks > 0 || needsLineBreak {
				replaceContent = f.eol
			} else {
				replaceContent = ""
			}
		default:
			switch firstToken {
			case OpenBrace:
				f.indentLevel++
				replaceContent = f.newLineAndIndent()
			case OpenBracket:
				f.indentLevel++
				replaceContent = f.newLineAndIndent()
			case Comma, LineComment:
				replaceContent = f.newLineAndIndent()
			case BlockComment:
				if f.numberLineBreaks > 0 {
					replaceContent = f.newLineAndIndent()
				} else if !needsLineBreak {
					replaceContent = " "
				}
			case Colon:
				replaceContent = " "
			case StringLiteral:
				if secondToken == Colon {
					replaceContent = ""
					break
				}
				fallthrough
			case NullKeyword, TrueKeyword, FalseKeyword, NumericLiteral,
				CloseBrace, CloseBracket:
				if secondToken == LineComment || secondToken == BlockComment {
					replaceContent = " "
				} else if secondToken != Comma && secondToken != Colon {
					// Two values with nothing between them is malformed input;
					// leave the document alone rather than guess.
					return
				}
			}
			if f.numberLineBreaks > 0 && needsLineBreak {
				replaceContent = f.newLineAndIndent()
			}
		}

		secondStart := f.s.Offset()
		if secondToken == EOF {
			secondStart = len(f.text)
		}
		f.addEdit(replaceContent, firstEnd, secondStart)

		firstToken = secondToken
	}
}
```

```go
package jsonc

import (
	"fmt"
	"sort"
	"strings"
)

// ApplyEdits applies edits to text and returns the result.
//
// Edits may be supplied in any order. They are applied back to front so
// offsets stay valid without bookkeeping. Two edits that overlap are an
// error rather than a silent last-write-wins; adjacent edits are fine.
func ApplyEdits(text string, edits []Edit) (string, error) {
	if len(edits) == 0 {
		return text, nil
	}

	// Copy so the caller's slice is not reordered.
	sorted := make([]Edit, len(edits))
	copy(sorted, edits)
	sort.SliceStable(sorted, func(i, j int) bool {
		if sorted[i].Offset != sorted[j].Offset {
			return sorted[i].Offset > sorted[j].Offset
		}
		return sorted[i].Length > sorted[j].Length
	})

	for i, e := range sorted {
		if e.Offset < 0 || e.Length < 0 || e.Offset > len(text) || e.Offset+e.Length > len(text) {
			return "", fmt.Errorf("jsonc: edit %+v is out of bounds for a document of length %d", e, len(text))
		}
		if i > 0 {
			// sorted is descending, so the previous edit starts later.
			prev := sorted[i-1]
			if e.Offset+e.Length > prev.Offset {
				return "", fmt.Errorf("jsonc: edits overlap: %+v and %+v", e, prev)
			}
		}
	}

	size := len(text)
	for _, e := range sorted {
		size += len(e.Content) - e.Length
	}
	if size < 0 {
		size = 0
	}

	var b strings.Builder
	b.Grow(size)
	// Walk forward for readability, using the ascending order.
	last := 0
	for i := len(sorted) - 1; i >= 0; i-- {
		e := sorted[i]
		b.WriteString(text[last:e.Offset])
		b.WriteString(e.Content)
		last = e.Offset + e.Length
	}
	b.WriteString(text[last:])
	return b.String(), nil
}
```

```go
type ModifyOptions struct {

	// surrounding document. A nil Format inserts compact text.

	// rather than replace it.

	// property names in document order. A nil func appends.
	GetInsertionIndex func(properties []string) int

func (o ModifyOptions) formatOptions() FormatOptions {
	if o.Format != nil {
		return *o.Format
	}
	return DefaultFormatOptions()

// Modify returns the edits that set the value at path.
//
// An empty path replaces the whole document. A path ending in a string names
// an object property, which is replaced if present and inserted otherwise; a
// path ending in an int names an array element (see Task 14).
//
// value is encoded with encoding/json, so any type with a marshaler works.
// Passing nil sets the JSON value null — it does not delete. Use Remove to
// delete.
func Modify(text string, path Path, value any, o ModifyOptions) ([]Edit, error) {
	encoded, err := json.Marshal(value)
	if err != nil {
		return nil, fmt.Errorf("jsonc: cannot encode value: %w", err)
	}
	return modify(text, path, string(encoded), false, o)
}

// modify is shared by Modify and Remove. content is the already-encoded
// replacement text; remove selects deletion, in which case content is ignored.
func modify(text string, path Path, content string, remove bool, o ModifyOptions) ([]Edit, error) {
	if err := validatePath(path); err != nil {
		return nil, err
	}

	root, _ := ParseTree(text)

	// Empty path: replace the whole document.
	if len(path) == 0 {
		if remove {
			return nil, fmt.Errorf("jsonc: cannot remove the document root")
		}
		if root == nil {
			return withFormatting(text, []Edit{{Offset: 0, Length: len(text), Content: content}}, o)
		}
		return withFormatting(text, []Edit{{Offset: root.Offset, Length: root.Length, Content: content}}, o)
	}

	last := path[len(path)-1]
	parentPath := path[:len(path)-1]
	parent := FindNodeAtLocation(root, parentPath)
	if parent == nil {
		return nil, fmt.Errorf("jsonc: cannot resolve path %v", parentPath)
	}

	switch key := last.(type) {
	case string:
		return modifyProperty(text, parent, key, content, remove, o)
	case int:
		return modifyElement(text, parent, key, content, remove, o)
	}
	return nil, fmt.Errorf("jsonc: unreachable")
}

func validatePath(path Path) error {
	for i, seg := range path {
		switch seg.(type) {
		case string, int:
		default:
			return fmt.Errorf("jsonc: path element %d is %T; must be string or int", i, seg)
		}
	}
	return nil
}

func modifyProperty(text string, parent *Node, key, content string, remove bool, o ModifyOptions) ([]Edit, error) {
	if parent.Type != NodeObject {
		return nil, fmt.Errorf("jsonc: cannot set property %q on a %s", key, parent.Type)
	}

	// Existing property: replace its value.
	for _, prop := range parent.Children {
		if len(prop.Children) != 2 {
			continue
		}
		k, ok := prop.Children[0].Literal.(string)
		if !ok || k != key {
			continue
		}
		if remove {
			return withFormatting(text, removalEdits(text, parent, prop), o)
		}
		v := prop.Children[1]
		return withFormatting(text, []Edit{{Offset: v.Offset, Length: v.Length, Content: content}}, o)
	}

	if remove {
		// Nothing to remove is not an error; it is already absent.
		return nil, nil
	}

	// New property: choose where it goes.
	names := make([]string, 0, len(parent.Children))
	for _, prop := range parent.Children {
		if len(prop.Children) > 0 {
			if k, ok := prop.Children[0].Literal.(string); ok {
				names = append(names, k)
			}
		}
	}
	index := len(names)
	if o.GetInsertionIndex != nil {
		index = o.GetInsertionIndex(names)
		if index < 0 {
			index = 0
		}
		if index > len(names) {
			index = len(names)
		}
	}

	encodedKey, err := json.Marshal(key)
	if err != nil {
		return nil, fmt.Errorf("jsonc: cannot encode property name: %w", err)
	}
	newText := string(encodedKey) + ": " + content

	return withFormatting(text, insertionEdits(parent, parent.Children, index, newText), o)
}

// insertionEdits produces the edit that splices newText in as the index'th
// child of parent, with the comma on the correct side.
func insertionEdits(parent *Node, children []*Node, index int, newText string) []Edit {
	if len(children) == 0 {
		// Empty container: insert just inside the opening delimiter.
		return []Edit{{Offset: parent.Offset + 1, Length: 0, Content: newText}}
	}
	if index >= len(children) {
		// Append: comma goes before the new text, after the last child.
		prev := children[len(children)-1]
		return []Edit{{Offset: prev.Offset + prev.Length, Length: 0, Content: "," + newText}}
	}
	// Insert before an existing child: comma goes after the new text.
	next := children[index]
	return []Edit{{Offset: next.Offset, Length: 0, Content: newText + ","}}
}

// removalEdits produces the edit that deletes child from parent, taking
// exactly one adjacent comma so the container stays well formed. The preceding
// comma is preferred, which is what keeps a trailing element's removal from
// leaving "{\"a\": 1, }".
//
// Adjacent comments are deliberately left in place: whether a comment belongs
// to the node before or after it cannot be decided from syntax, and deleting a
// user's comment is worse than leaving an orphan.
func removalEdits(text string, parent, child *Node) []Edit {
	start, end := child.Offset, child.Offset+child.Length

	// Find this child's index among its siblings.
	idx := -1
	for i, c := range parent.Children {
		if c == child {
			idx = i
			break
		}
	}

	if idx > 0 {
		// Extend backward through whitespace to the preceding comma.
		i := start - 1
		for i >= 0 && isSpaceByte(text[i]) {
			i--
		}
		if i >= 0 && text[i] == ',' {
			start = i
		}
	} else if idx == 0 && len(parent.Children) > 1 {
		// First of several: take the comma that follows.
		i := end
		for i < len(text) && isSpaceByte(text[i]) {
			i++
		}
		if i < len(text) && text[i] == ',' {
			end = i + 1
		}
	}

	return []Edit{{Offset: start, Length: end - start, Content: ""}}
}

func isSpaceByte(c byte) bool {
	return c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\v' || c == '\f'
}

// withFormatting applies edits, formats the affected range of the result, and
// returns a single edit set expressed against the ORIGINAL text.
//
// Returning one edit covering the touched range keeps the result simple and
// guarantees the caller never sees overlapping edits.
func withFormatting(text string, edits []Edit, o ModifyOptions) ([]Edit, error) {
	if len(edits) == 0 {
		return nil, nil
	}
	if o.Format == nil {
		return edits, nil
	}

	modified, err := ApplyEdits(text, edits)
	if err != nil {
		return nil, err
	}

	// The span of the modified document that the edits touched.
	start := edits[0].Offset
	end := edits[0].Offset + edits[0].Length
	delta := 0
	for _, e := range edits {
		if e.Offset < start {
			start = e.Offset
		}
		if e.Offset+e.Length > end {
			end = e.Offset + e.Length
		}
		delta += len(e.Content) - e.Length
	}
	newEnd := end + delta

	fmtEdits := Format(modified, &Range{Offset: start, Length: newEnd - start}, o.formatOptions())
	formatted, err := ApplyEdits(modified, fmtEdits)
	if err != nil {
		return nil, err
	}

	return diffEdit(text, formatted), nil
}

// diffEdit expresses the difference between two strings as a single edit
// covering the span between their common prefix and common suffix.
func diffEdit(before, after string) []Edit {
	if before == after {
		return nil
	}
	p := 0
	for p < len(before) && p < len(after) && before[p] == after[p] {
		p++
	}
	sb, sa := len(before), len(after)
	for sb > p && sa > p && before[sb-1] == after[sa-1] {
		sb--
		sa--
	}
	return []Edit{{Offset: p, Length: sb - p, Content: after[p:sa]}}
}
```

```go
func modifyElement(text string, parent *Node, index int, content string, remove bool, o ModifyOptions) ([]Edit, error) {
	if parent.Type != NodeArray {
		return nil, fmt.Errorf("jsonc: cannot index a %s", parent.Type)
	}
	n := len(parent.Children)

	if remove {
		if index < 0 || index >= n {
			return nil, nil // already absent
		}
		return withFormatting(text, removalEdits(text, parent, parent.Children[index]), o)
	}

	// -1 or past the end appends.
	if index < 0 || index >= n {
		if index >= 0 && index > n {
			return nil, fmt.Errorf("jsonc: index %d is out of range for an array of length %d", index, n)
		}
		return withFormatting(text, insertionEdits(parent, parent.Children, n, content), o)
	}

	if o.IsArrayInsertion {
		return withFormatting(text, insertionEdits(parent, parent.Children, index, content), o)
	}

	elem := parent.Children[index]
	return withFormatting(text, []Edit{{Offset: elem.Offset, Length: elem.Length, Content: content}}, o)
}
```

```go
// Remove returns the edits that delete the value at path, along with exactly
// one adjacent comma so the enclosing container stays well formed.
//
// Removing something that is not there is not an error; it returns no edits.
// Removing the document root is an error.
//
// Comments next to the removed value are left in place. Deciding whether a
// comment belongs to the value before or after it is not possible from syntax
// alone, and deleting a user's comment silently is worse than leaving an
// orphan. node-jsonc-parser behaves the same way.
func Remove(text string, path Path, o ModifyOptions) ([]Edit, error) {
	return modify(text, path, "", true, o)
```

```go
func FuzzParse(f *testing.F) {
	seeds(f)
	f.Fuzz(func(t *testing.T, text string) {
		v, errs := Parse(text)

		for _, e := range errs {
			if e.Offset < 0 || e.Offset > len(text) {
				t.Fatalf("error offset %d out of bounds for len %d: %+v", e.Offset, len(text), e)
			}
			if e.Length < 0 || e.Offset+e.Length > len(text) {
				t.Fatalf("error span %d+%d out of bounds for len %d: %+v", e.Offset, e.Length, len(text), e)
			}
			if e.StartLine < 0 || e.StartCol < 0 {
				t.Fatalf("negative line/col: %+v", e)
			}
		}

		// Where the input is also valid strict JSON, agree with the stdlib.
		var want any
		if json.Unmarshal([]byte(text), &want) == nil && len(errs) == 0 {
			if !reflect.DeepEqual(v, want) {
				t.Fatalf("Parse(%q) = %#v, encoding/json = %#v", text, v, want)
			}
		}
	})
}

func FuzzScanner(f *testing.F) {
	seeds(f)
	f.Fuzz(func(t *testing.T, text string) {
		s := NewScanner(text)
		prevEnd := 0
		for {
			k := s.Scan()
			if k == EOF {
				break
			}
			off, length := s.Offset(), s.Length()
			if off < 0 || length < 0 || off+length > len(text) {
				t.Fatalf("token %v span %d+%d out of bounds for len %d", k, off, length, len(text))
			}
			if off != prevEnd {
				t.Fatalf("token %v starts at %d, previous ended at %d: tokens must tile the input", k, off, prevEnd)
			}
			if length == 0 {
				t.Fatalf("token %v has zero length at %d: scanner would not advance", k, off)
			}
			prevEnd = off + length
			if s.StartColumn() < 0 {
				t.Fatalf("negative column at offset %d", off)
			}
		}
		if prevEnd != len(text) {
			t.Fatalf("tokens covered %d of %d bytes", prevEnd, len(text))
		}
	})
}

func FuzzFormatIdempotent(f *testing.F) {
	seeds(f)
	f.Fuzz(func(t *testing.T, text string) {
		o := FormatOptions{TabSize: 2, InsertSpaces: true, EOL: "\n"}

		before, errs := Parse(text)
		if len(errs) != 0 {
			return // the invariant is defined only for clean input
		}

		once, err := ApplyEdits(text, Format(text, nil, o))
		if err != nil {
			t.Fatalf("Format produced unapplicable edits for %q: %v", text, err)
		}

		after, errs2 := Parse(once)
		if len(errs2) != 0 {
			t.Fatalf("formatting %q produced %q, which does not parse: %v", text, once, errs2)
		}
		if !reflect.DeepEqual(before, after) {
			t.Fatalf("formatting changed the value of %q:\n before %#v\n after  %#v", text, before, after)
		}

		if edits := Format(once, nil, o); len(edits) != 0 {
			t.Fatalf("formatting %q twice was not idempotent; second pass: %+v", text, edits)
		}
	})
}

func FuzzModifyApply(f *testing.F) {

	f.Fuzz(func(t *testing.T, text string) {
		if _, errs := Parse(text); len(errs) != 0 {
			return
		}
		root, _ := ParseTree(text)
		if root == nil || root.Type != NodeObject {
			return
		}

		o := FormatOptions{TabSize: 2, InsertSpaces: true, EOL: "\n"}
		edits, err := Modify(text, Path{"fuzzKey"}, 42, ModifyOptions{Format: &o})
		if err != nil {
			t.Fatalf("Modify(%q): %v", text, err)
		}
		out, err := ApplyEdits(text, edits)
		if err != nil {
			t.Fatalf("applying edits to %q: %v", text, err)

		if len(errs) != 0 {
			t.Fatalf("Modify(%q) produced %q, which does not parse: %v", text, out, errs)
		}
		m, ok := v.(map[string]any)
		if !ok {
			t.Fatalf("result of modifying an object is %T, want map", v)
		}
		if m["fuzzKey"] != float64(42) {
			t.Fatalf("after Modify, fuzzKey = %#v, want 42; document: %q", m["fuzzKey"], out)
		}
	})
```
